1//! # Credentials Module
2//!
3//! This module provides data structures and functionality for handling user credentials,
4//! including password hashing, verification, and secure management of user authentication data.
5//!
6//! ## Overview
7//!
8//! - Defines the [`Credentials`] struct, which encapsulates a user's authentication information.
9//! - Provides methods for creating credentials from plaintext passwords or precomputed hashes.
10//! - Supports password verification using pluggable password managers.
11//!
12//! ## Features
13//!
14//! - Secure password hashing and verification
15//! - Extensible password management via the [`SecurePasswordManager`] trait
16//! - Designed for use with unique user identifiers (UUID, email, username)
17//!
18//! ## Modules
19//!
20//! - [`plain_password`]: Contains the [`PlainPassword`] type for handling plaintext passwords.
21//!
22//! ## Example
23//!
24//! ```rust,no_run
25//! use authen::core::credentials::{Credentials, PlainPassword};
26//! use authen::core::password::SecurePasswordManager;
27//! use authen::error::AuthError;
28//! # struct DummyManager;
29//! # #[async_trait::async_trait]
30//! # impl SecurePasswordManager for DummyManager {
31//! #     async fn hash_password(&self, password: &str) -> Result<String, AuthError> { Ok(password.to_owned()) }
32//! #     async fn verify_password(&self, password: &str, hash: &str) -> Result<bool, AuthError> { Ok(password == hash) }
33//! # }
34//! # tokio::runtime::Runtime::new().unwrap().block_on(async {
35//! let manager = DummyManager;
36//! let plain = PlainPassword::new("my_password".to_string());
37//! let creds = Credentials::from_plain_password(&manager, "user-1".to_string(), "user@example.com".to_string(), plain).await.unwrap();
38//! assert!(creds.verify_password(&manager, &PlainPassword::new("my_password".to_string())).await.unwrap());
39//! # });
40//! ```

46/// Represents a user's credentials, including identifiers and hashed password.
47///
48/// This struct is used to store and manage authentication data for a user.
49///
50/// # Fields
51///
52/// - `user_id`: Unique identifier for the user (preferably a UUID).
53/// - `identifier`: Unique user identifier (such as email or username).
54/// - `password_hash`: Securely hashed password.
55#[derive(Debug, Clone, Default)]
56pub struct Credentials {
57    /// Unique identifier for the user (preferably UUID)
58    pub user_id: String,
59    /// Unique identifier for the user (preferably email or username)
60    pub identifier: String,
61    /// Hashed password
62    pub password_hash: String,
63}
64
65impl Credentials {
66    /// Creates a new [`Credentials`] instance from an already hashed password.
67    ///
68    /// # Arguments
69    ///
70    /// * `user_id` - Unique identifier for the user (e.g., UUID).
71    /// * `identifier` - User's login identifier (e.g., email or username).
72    /// * `password_hash` - Precomputed password hash.
73    ///
74    /// # Returns
75    ///
76    /// A new [`Credentials`] object containing the provided data.
77    pub fn new(user_id: String, identifier: String, password_hash: String) -> Self {
78        Self {
79            user_id,
80            identifier,
81            password_hash,
82        }
83    }
84
85    /// Creates credentials by hashing a plaintext password using the provided password manager.
86    ///
87    /// # Arguments
88    ///
89    /// * `manager` - Reference to a type implementing [`SecurePasswordManager`] for hashing.
90    /// * `user_id` - Unique identifier for the user (e.g., UUID).
91    /// * `identifier` - User's login identifier (e.g., email or username).
92    /// * `plain_password` - The user's plaintext password.
93    ///
94    /// # Errors
95    ///
96    /// Returns [`AuthError::HashingError`] if password hashing fails.
97    ///
98    /// # Returns
99    ///
100    /// A [`Credentials`] object with the hashed password on success.
101    pub async fn from_plain_password(
102        manager: &(dyn crate::core::password::SecurePasswordManager + Send + Sync),
103        user_id: String,
104        identifier: String,
105        plain_password: PlainPassword,
106    ) -> Result<Self, crate::error::AuthError> {
107        let password_hash = manager
108            .hash_password(plain_password.as_str())
109            .await
110            .map_err(|e| crate::error::AuthError::HashingError(format!("Couldn't hash : {e}")))?;
111
112        Ok(Self {
113            user_id,
114            identifier,
115            password_hash,
116        })
117    }
118
119    /// Verifies a plaintext password against the stored password hash.
120    ///
121    /// # Arguments
122    ///
123    /// * `manager` - Reference to a type implementing [`SecurePasswordManager`] for verification.
124    /// * `plain_password` - The plaintext password to verify.
125    ///
126    /// # Errors
127    ///
128    /// Returns [`AuthError::VerificationError`] if verification fails.
129    ///
130    /// # Returns
131    ///
132    /// `Ok(true)` if the password matches, `Ok(false)` if it does not, or an error if verification fails.
133    pub async fn verify_password(
134        &self,
135        manager: &(dyn crate::core::password::SecurePasswordManager + Send + Sync),
136        plain_password: &PlainPassword,
137    ) -> Result<bool, crate::error::AuthError> {
138        manager
139            .verify_password(plain_password.as_str(), &self.password_hash)
140            .await
141            .map_err(|e| {
142                crate::error::AuthError::VerificationError(format!("Couldn't verify : {e}"))
143            })
144    }
145}
