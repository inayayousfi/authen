1//! In-memory implementation of the `UserRepository` trait for testing and development.
2//!
3//! This module provides a thread-safe, in-memory user repository using `Arc<Mutex<Vec<User>>>`.
4//! It is intended for use in tests or non-persistent environments where a database is not required.
5
6use async_trait::async_trait;
7
8use super::traits::UserRepository;
9use crate::core::user::User;
10use std::sync::{Arc, Mutex};
11
12/// Thread-safe, in-memory implementation of the [`UserRepository`] trait.
13///
14/// Stores users in a shared, mutable vector protected by a mutex.
15/// Suitable for testing or ephemeral use cases where persistence is not required.
16#[derive(Default, Debug)]
17pub struct InMemoryUserRepo {
18    /// Shared, thread-safe vector of users.
19    users: Arc<Mutex<Vec<User>>>,
20}
21
22impl InMemoryUserRepo {
23    /// Creates a new, empty in-memory user repository.
24    ///
25    /// # Examples
26    ///
27    /// ```rust,no_run
28    /// use authen::core::user::persistence::InMemoryUserRepo;
29    /// let repo = InMemoryUserRepo::new();
30    /// ```
31    pub fn new() -> Self {
32        InMemoryUserRepo {
33            users: Arc::new(Mutex::new(Vec::new())),
34        }
35    }
36}
37
38#[async_trait]
39impl UserRepository for InMemoryUserRepo {
40    /// Adds a new user to the repository.
41    ///
42    /// # Arguments
43    /// * `user` - The user to add.
44    ///
45    /// # Returns
46    /// * `Ok(User)` if the user was added successfully.
47    /// * `Err(AuthError)` if the repository is unavailable.
48    async fn add_user(&self, user: User) -> Result<User, crate::error::AuthError> {
49        let mut users = self
50            .users
51            .lock()
52            .map_err(|e| crate::error::AuthError::ServiceUnavailable(e.to_string()))?;
53        users.push(user.clone());
54        Ok(user.clone())
55    }
56
57    /// Retrieves a user by their unique ID.
58    ///
59    /// # Arguments
60    /// * `id` - The user's unique identifier.
61    ///
62    /// # Returns
63    /// * `Some(User)` if found, or `None` if not found or on lock error.
64    async fn get_user_by_id(&self, id: &str) -> Option<User> {
65        let users = self.users.lock().ok()?; // Handle potential poisoning
66        users.iter().find(|u| u.id == id).cloned()
67    }
68
69    /// Retrieves a user by their identifier (e.g., username or email).
70    ///
71    /// # Arguments
72    /// * `identifier` - The user's identifier.
73    ///
74    /// # Returns
75    /// * `Some(User)` if found, or `None` if not found or on lock error.
76    async fn get_user_by_identifier(&self, identifier: &str) -> Option<User> {
77        let users = self.users.lock().ok()?; // Handle potential poisoning
78        users
79            .iter()
80            .find(|u| {
81                u.credentials
82                    .as_ref()
83                    .map(|creds| creds.identifier == identifier)
84                    .unwrap_or(false)
85            })
86            .cloned()
87    }
88
89    /// Updates an existing user in the repository.
90    ///
91    /// # Arguments
92    /// * `user` - The user with updated information.
93    ///
94    /// # Returns
95    /// * `Ok(())` if the user was updated.
96    /// * `Err(AuthError::UserNotFound)` if the user does not exist.
97    /// * `Err(AuthError)` if the repository is unavailable.
98    async fn update_user(&self, user: &User) -> Result<(), crate::error::AuthError> {
99        let mut users = self
100            .users
101            .lock()
102            .map_err(|e| crate::error::AuthError::ServiceUnavailable(e.to_string()))?;
103        if let Some(existing) = users.iter_mut().find(|u| u.id == user.id) {
104            *existing = user.clone();
105            Ok(())
106        } else {
107            Err(crate::error::AuthError::UserNotFound)
108        }
109    }
110
111    /// Deletes a user from the repository by their ID.
112    ///
113    /// # Arguments
114    /// * `id` - The user's unique identifier.
115    ///
116    /// # Returns
117    /// * `Ok(())` if the user was deleted.
118    /// * `Err(AuthError::UserNotFound)` if the user does not exist.
119    /// * `Err(AuthError)` if the repository is unavailable.
120    async fn delete_user(&self, id: &str) -> Result<(), crate::error::AuthError> {
121        let mut users = self
122            .users
123            .lock()
124            .map_err(|e| crate::error::AuthError::ServiceUnavailable(e.to_string()))?;
125        let len_before = users.len();
126        users.retain(|u| u.id != id);
127        if users.len() < len_before {
128            Ok(())
129        } else {
130            Err(crate::error::AuthError::UserNotFound)
131        }
132    }
133
134    /// Retrieves a user by their OAuth provider and provider user ID.
135    ///
136    /// # Arguments
137    /// * `provider` - The OAuth2 provider.
138    /// * `provider_user_id` - The user ID from the OAuth provider.
139    ///
140    /// # Returns
141    /// * `Some(User)` if found, or `None` if not found or on lock error.
142    async fn get_user_by_oauth_id(
143        &self,
144        provider: crate::core::oauth::store::OAuth2Provider,
145        provider_user_id: &str,
146    ) -> Option<User> {
147        let users = self.users.lock().ok()?;
148        users
149            .iter()
150            .find(|u| {
151                u.oauth_accounts
152                    .get(&provider)
153                    .map(|oauth_info| oauth_info.provider_user_id == provider_user_id)
154                    .unwrap_or(false)
155            })
156            .cloned()
157    }
158}
