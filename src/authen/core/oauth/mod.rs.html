1//! OAuth2 core module for authentication and authorization.
2//!
3//! This module provides abstractions and implementations for handling OAuth2 flows,
4//! including generating authorization URLs, exchanging codes for tokens, fetching user info,
5//! and refreshing tokens. It is designed to support multiple providers and extensible flows.
6//!
7//! # Modules
8//!
9//! - `manager`: Contains the logic for managing OAuth2 operations and provider-specific details.
10//! - `store`: Defines types and storage mechanisms for OAuth2 tokens, user info, and providers.
11//!
12//! # Traits
13//!
14//! - [`OAuth2Service`]: The main trait for interacting with OAuth2 providers.
15
16use async_trait::async_trait;

20/// Trait for handling OAuth2 authentication and authorization flows.
21///
22/// Implementors of this trait provide methods for generating authorization URLs,
23/// exchanging authorization codes for tokens, fetching user information, and refreshing tokens.
24/// This trait is designed to be provider-agnostic and extensible for various OAuth2 providers.
25///
26/// # Example
27///
28/// ```rust,ignore
29/// let url = service.generate_auth_url(provider, state, Some(vec!["email".to_string()])).await?;
30/// ```
31pub trait OAuth2Service {
32    /// Generates an authorization URL for the specified OAuth2 provider.
33    ///
34    /// This method constructs the URL to which users should be redirected to begin the OAuth2 flow.
35    /// It includes the necessary parameters for CSRF protection and can request additional scopes.
36    ///
37    /// # Arguments
38    ///
39    /// * `provider` - The OAuth2 provider for which to generate the authorization URL.
40    /// * `state` - A unique state string for CSRF protection.
41    /// * `scopes` - Optional list of additional scopes to request beyond the provider's defaults.
42    ///
43    /// # Returns
44    ///
45    /// * `Ok(String)` - The generated authorization URL.
46    /// * `Err(AuthError)` - If URL generation fails due to configuration or provider errors.
47    async fn generate_auth_url(
48        &self,
49        provider: store::OAuth2Provider,
50        state: &str,
51        scopes: Option<Vec<String>>,
52    ) -> Result<String, crate::AuthError>;
53
54    /// Exchanges an authorization code for an access token with the OAuth2 provider.
55    ///
56    /// This method is called after the user has authorized the application and the provider
57    /// has redirected back with an authorization code. It verifies the state and requests
58    /// an access token from the provider.
59    ///
60    /// # Arguments
61    ///
62    /// * `provider` - The OAuth2 provider to exchange the code with.
63    /// * `code` - The authorization code received from the provider.
64    /// * `state` - The state parameter for CSRF protection and verification.
65    ///
66    /// # Returns
67    ///
68    /// * `Ok(OAuth2Token)` - The access token and related information.
69    /// * `Err(AuthError)` - If the exchange fails due to invalid code, state, or provider error.
70    async fn exchange_code_for_token(
71        &self,
72        provider: store::OAuth2Provider,
73        code: &str,
74        state: &str,
75    ) -> Result<store::OAuth2Token, crate::AuthError>;
76
77    /// Fetches user information from the OAuth2 provider using the provided access token.
78    ///
79    /// This method retrieves user profile data from the provider's user info endpoint.
80    ///
81    /// # Arguments
82    ///
83    /// * `token` - The OAuth2 token to use for authentication when fetching user info.
84    ///
85    /// # Returns
86    ///
87    /// * `Ok(OAuth2UserInfo)` - The user's profile information as returned by the provider.
88    /// * `Err(AuthError)` - If fetching user info fails due to invalid token or provider error.
89    async fn fetch_user_info(
90        &self,
91        token: &store::OAuth2Token,
92    ) -> Result<store::OAuth2UserInfo, crate::AuthError>;
93
94    /// Refreshes an access token using a refresh token.
95    ///
96    /// This method requests a new access token from the provider using a valid refresh token.
97    /// Useful for maintaining long-lived sessions without requiring user re-authentication.
98    ///
99    /// # Arguments
100    ///
101    /// * `token` - The OAuth2 token containing the refresh token.
102    ///
103    /// # Returns
104    ///
105    /// * `Ok(OAuth2Token)` - The new access token and related information.
106    /// * `Err(AuthError)` - If refreshing fails due to invalid token or provider error.
107    async fn refresh_token(
108        &self,
109        token: &store::OAuth2Token,
110    ) -> Result<store::OAuth2Token, crate::AuthError>;
111
112    /// Gets the redirect_frontend_uri for the given OAuth2 provider.
113    ///
114    /// This method returns the frontend redirect URI that should be used to redirect
115    /// users back to the frontend application after OAuth2 completion.
116    ///
117    /// # Arguments
118    ///
119    /// * `provider` - The OAuth2 provider to get the redirect_frontend_uri for.
120    ///
121    /// # Returns
122    ///
123    /// * `Ok(String)` - The redirect_frontend_uri as configured for the provider.
124    /// * `Err(AuthError)` - If the provider configuration is missing or invalid.
125    async fn get_redirect_frontend_uri(
126        &self,
127        provider: store::OAuth2Provider,
128    ) -> Result<String, crate::AuthError>;
129}
