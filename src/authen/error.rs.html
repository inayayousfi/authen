1use thiserror::Error;

3/// All possible errors that can occur within the authentication crate.
4///
5/// This enum is used throughout the crate to represent error conditions that may arise
6/// during authentication, user management, password handling, token operations, and more.
7/// Each variant is documented to clarify the context in which it is used.
8
9#[derive(Debug, Error)]
10pub enum AuthError {
11    /// Returned when the provided credentials are invalid (e.g., wrong password or username).
12    #[error("Invalid credentials provided.")]
13    InvalidCredentials,
14
15    /// Returned when a user cannot be found in the data store.
16    #[error("User not found.")]
17    UserNotFound,
18
19    /// Returned when attempting to create a user that already exists.
20    #[error("User already exists.")]
21    UserAlreadyExists,
22
23    /// Returned when password hashing fails.
24    /// Contains the underlying error message.
25    #[error("Password hashing failed: {0}")]
26    HashingError(String),
27
28    /// Returned when password verification fails.
29    /// Contains the underlying error message.
30    #[error("Password verification failed: {0}")]
31    PasswordVerificationError(String),
32
33    /// Returned when there is a configuration error in the authentication system.
34    /// Contains a description of the configuration issue.
35    #[error("Configuration error: {0}")]
36    ConfigError(String),
37
38    /// Returned when a service required for authentication is unavailable.
39    /// Contains a description of the service issue.
40    #[error("Service unavailable: {0}")]
41    ServiceUnavailable(String),
42
43    /// Returned when a feature is not yet implemented.
44    /// Contains a description of the missing feature.
45    #[error("Feature not implemented yet: {0}")]
46    NotImplemented(String),
47
48    /// Returned when input data is invalid or malformed.
49    /// Contains a description of the invalid input.
50    #[error("Invalid input data: {0}")]
51    InvalidInput(String),
52
53    /// Returned when a password does not meet policy requirements or is otherwise invalid.
54    /// Contains a description of the password issue.
55    #[error("Invalid password: {0}")]
56    InvalidPassword(String),
57
58    /// Returned when a verification process fails (e.g., email or token verification).
59    /// Contains a description of the verification error.
60    #[error("Verification error: {0}")]
61    VerificationError(String),
62
63    /// Returned when a token (such as a JWT) has expired and is no longer valid.
64    #[error("Token expired")]
65    TokenExpired,
66
67    /// Returned when token generation fails.
68    /// Contains a description of the generation error.
69    #[error("Token generation failed: {0}")]
70    TokenGeneration(String),
71
72    /// Returned when token validation fails.
73    /// Contains a description of the validation error.
74    #[error("Token validation failed: {0}")]
75    TokenValidation(String),
76
77    /// Returned when a token is invalid (malformed, tampered, or otherwise unusable).
78    /// Contains a description of the invalid token.
79    #[error("InvalidToken: {0}")]
80    InvalidToken(String),
81
82    /// Returned when the password manager component is missing or unavailable.
83    #[error("Missing password manager")]
84    MissingPasswordManager,
85
86    /// Returned when the persistent user manager component is missing or unavailable.
87    #[error("Missing persistent user manager")]
88    MissingPersistentUserManager,
89
90    /// Returned when the token manager component is missing or unavailable.
91    #[error("Missing token manager")]
92    MissingTokenManager,
93    /// Returned when a network error occurs during OAuth operations.
94    #[error("OAuth network error: {0}")]
95    OAuthNetwork(String),
96
97    /// Returned when an invalid response is received from the OAuth provider.
98    #[error("OAuth invalid response: {0}")]
99    OAuthInvalidResponse(String),
100
101    /// Returned when the OAuth provider returns an error.
102    #[error("OAuth provider error: {0}")]
103    OAuthProvider(String),
104
105    /// Returned when the token exchange process fails during OAuth.
106    #[error("OAuth token exchange failed: {0}")]
107    OAuthTokenExchange(String),
108
109    /// Returned when fetching user info from the OAuth provider fails.
110    #[error("OAuth user info fetch failed: {0}")]
111    OAuthUserInfo(String),
112
113    /// Returned for other errors related to OAuth operations.
114    #[error("OAuth other error: {0}")]
115    OAuthOther(String),
116
117    /// Returned when a user signup operation fails.
118    /// Contains a description of the signup error.
119    #[error("User signup error: {0}")]
120    SignupError(String),
121
122    /// Returned when a user login operation fails.
123    /// Contains a description of the login error.
124    #[error("User login error: {0}")]
125    LoginError(String),
126
127    /// Returned when a database error occurs (only available with the `postgres` feature).
128    /// Contains a description of the database error.
129    #[cfg(feature = "postgres")]
130    #[error("Database error: {0}")]
131    DatabaseError(String),
132}
